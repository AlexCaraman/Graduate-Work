regions = ["'Baltimore-Columbia-Towson, MD'","'California-Lexington Park, MD'","'Cambridge, MD'","'Cumberland, MD-WV'",
           "'Easton, MD'","'Hagerstown-Martinsburg, MD-WV'","'Philadelphia-Camden-Wilmington, PA-NJ-DE-MD'",
           "'Salisbury, MD-DE'","'Washington-Arlington-Alexandria, DC-VA-MD-WV'"]
per_block_by_region = {}
stdev_region = [] 
means_region = mean_jobs_region.iloc[:,2]
for name in regions:
    per_block_by_region[name] = sql('SELECT c000 \
                                     FROM public.md_wac_2019 JOIN public.md_crosswalk2 \
                                     ON md_wac_2019.w_geocode = md_crosswalk2.tabblk2010 \
                                     WHERE cbsaname={}'.format(name))
    region_disp = np.std(per_block_by_region[name]) 
    stdev_region.append(region_disp)
    print('The standard deviations of total jobs in {} is {}'.format(name,float(region_disp)))
