def get_state_data(state,year1,year2,characteristic):
    # state: USA state of interest, a string
    # year1, year2: first and last year of data of interest, either both integers or one None
    # characteristic: Workplace or Residence area (WAC or RAC), or origin-destination (OD), a string
    data_dict = {}
    char = characteristic.lower() # convert to lower-case to avoid case-sensitive errors
    state = state.lower()
    char_options = ['wac', 'rac']
    if len(state) == 2:
        # for loading single data set 
        if year2 is None:
            if char in char_options:
                file = 'https://lehd.ces.census.gov/data/lodes/LODES7/{place}/{cha}/{place}_{cha}_S000_JT00_{yr}.csv.gz'.format(place=state,cha=char,yr=year1)
                data_set = pd.read_csv(file,compression='gzip')
                return(data_set)
            elif char =='od':
                file = 'https://lehd.ces.census.gov/data/lodes/LODES7/{place}/{cha}/{place}_{cha}_aux_JT00_{yr}.csv.gz'.format(place=state,cha=char,yr=year1)
                data_set = pd.read_csv(file,compression='gzip')
                return(data_set)
            else:
                print('Please choose between WAC, RAC, or Origin-Destination (OD) data.') 
                
        # in this case, ensure year2 is an integer and can be used to generate dictionary         
        elif not isinstance(year2,int) or year2<=year1:
            print('If specified, please ensure the second year parameter is an integer greater than the first year parameter.')
            
        # for loading multiple data sets, create a dictionary
        elif isinstance(year2,int):
            if char in char_options:
                for i in range(year1,year2+1):
                    file = 'https://lehd.ces.census.gov/data/lodes/LODES7/{place}/{cha}/{place}_{cha}_S000_JT00_{yr}.csv.gz'.format(place=state,cha=char,yr=i)
                    data_set = pd.read_csv(file,compression='gzip')
                    data_dict[i] = data_set
                return(data_dict)
            elif char =='od':
                for i in range(year1, year2+1):
                    file = 'https://lehd.ces.census.gov/data/lodes/LODES7/{place}/{cha}/{place}_{cha}_aux_JT00_{yr}.csv.gz'.format(place=state,cha=char,yr=i)
                    data_set = pd.read_csv(file,compression='gzip')
                    data_dict[i] = data_set
                return(data_dict)
            else:
                print('Please choose between WAC, RAC, or Origin-Destination (OD) data.')
    else:
        print('Please specify the state by its two-letter abbreviation.')

          'Wholesale','Retail','Transportation/Warehousing','Information','Finance/Insurance',
          'Real Estate','Prof/Scientific/Tech Services','Management','Waste Management','Education',
          'Health Care','Arts/Entertainment/Recreation','Accomodation/Food','Other','Public Admin']
for sector in range(10,29+1):
    industry_proportions[sector] = []
for year in range(2002,2016+1):
    for sector in range(10,29+1):
        total_sector = np.sum(md_data[year].iloc[:,sector])
        total_jobs = np.sum(md_data[year].iloc[:,1])
        proportion = total_sector/total_jobs
        industry_proportions[sector].append(proportion)
        
for i in range(0,19+1):
